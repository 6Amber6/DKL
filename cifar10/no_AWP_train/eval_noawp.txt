# AutoAttack 评估四个 noAWP 方法 (epoch 200, 180, 160)
# 在 cifar10/ 目录下运行，checkpoint 在 cifar10/ 下
#
# 运行前设置:
#   cd /home/tong.li003/DKL/cifar10
#   export PYTHONPATH=/home/tong.li003/DKL:$PYTHONPATH
#   mkdir -p logs

# ========== 1. parallel ==========
for ep in 200 180 160; do
  python -m cifar10.eval.parallel \
    --checkpoint cifar10_parallel_noawp_a4b20t4_s6/fusion-epoch${ep}.pt \
    --data CIFAR10 \
    --preprocess '01' \
    --log_path logs/cifar10_parallel_noawp_a4b20t4_s6_epoch${ep}.log
done

# ========== 2. gated ==========
for ep in 200 180 160; do

    --log_path logs/cifar10_gated_noawp_a4b20t4_s6_epoch${ep}.log
