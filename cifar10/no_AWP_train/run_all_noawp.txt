# Run four noAWP training methods (no AWP, EMA)
# Run from DKL project root. Set: PROJECT_ROOT=/path/to/DKL, ADV_DIR=$PROJECT_ROOT/DKLv1/Adv-training-dkl
# cd $PROJECT_ROOT && export PYTHONPATH=$PROJECT_ROOT:$ADV_DIR:$PYTHONPATH

# 1. parallel
python cifar10/no_AWP_train/noAWP_parallel.py \
  --epochs-sub 100 \
  --epochs-fusion 200 \
  --batch-size 128 \
  --lr 0.1 \
  --epsilon 0.031372549 \
  --alpha 4.0 \
  --beta 20.0 \
  --T 4.0 \
  --train_budget high \
  --model-dir $ADV_DIR/workdir \
  --mark cifar10_parallel_noawp_a4b20t4_s6 \
  --data-path $ADV_DIR/../data \
  --save-start 100 \
  --save-freq 20 \
  --seed 0
