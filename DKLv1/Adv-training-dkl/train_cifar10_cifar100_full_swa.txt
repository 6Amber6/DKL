# Reproduce paper: train + swa.py (EMA over epoch 1-200)
# save_freq=1 -> ~70 GB per exp, ~140 GB for both

# CIFAR-10 
# train (run in DKLv1/Adv-training-dkl)
python train_dkl_cifar10.py \
  --arch WideResNet34_10 \
  --data CIFAR10 \
  --train_budget 'high' \
  --mark cifar10_dkl_a4b20t4 \
  --epsilon 8 \
  --lr 0.2 \
  --beta 20.0 \
  --alpha 4.0 \
  --T 4.0 \
  --epochs 200 \
  --seed 0

# swa (run after train, still in DKLv1/Adv-training-dkl)
python swa.py workdir/cifar10_dkl_a4b20t4 0.9 1 200
